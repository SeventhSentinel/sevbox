// Thanks to Vortex who provided the base version of this script

version "4.10"

class SevBoxHandler : EventHandler
{
	TextureID frames[8];
	int currentFrame;
	int tickCounter;
	bool active;

	const CYCLE_TICS = 3;

	override void WorldLoaded(WorldEvent e)
	{
		frames[0] = TexMan.CheckForTexture("CJSky3A", TexMan.Type_Any);
		frames[1] = TexMan.CheckForTexture("CJSky3B", TexMan.Type_Any);
		frames[2] = TexMan.CheckForTexture("CJSky3C", TexMan.Type_Any);
		frames[3] = TexMan.CheckForTexture("CJSky3D", TexMan.Type_Any);
		frames[4] = TexMan.CheckForTexture("CJSky3E", TexMan.Type_Any);
		frames[5] = TexMan.CheckForTexture("CJSky3F", TexMan.Type_Any);
		frames[6] = TexMan.CheckForTexture("CJSky3G", TexMan.Type_Any);
		frames[7] = TexMan.CheckForTexture("CJSky3H", TexMan.Type_Any);

		TextureID sky3a = TexMan.CheckForTexture("CJSky3A", TexMan.Type_Any);
		active = (level.skytexture1 == sky3a || level.skytexture1 == frames[0]);
		currentFrame = 0;
		tickCounter = 0;
	}

	override void WorldTick()
	{
		if (!active) return;
		tickCounter++;
		if (tickCounter >= CYCLE_TICS)
		{
			tickCounter = 0;
			currentFrame = (currentFrame + 1) % 8;
			level.ChangeSky(frames[currentFrame], frames[currentFrame]);
		}
	}
}
